-- Listas de compra (una por cada vez que se va a comprar, para no mezclar compras distintas)
CREATE TABLE compras_listas (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id uuid REFERENCES auth.users NOT NULL,
  nombre text NOT NULL,
  fecha date NOT NULL DEFAULT current_date,
  estado text NOT NULL DEFAULT 'abierta' CHECK (estado IN ('abierta','cerrada')),
  es_compartida boolean DEFAULT false,
  created_at timestamptz DEFAULT now(),
  closed_at timestamptz
);
ALTER TABLE compras_listas ENABLE ROW LEVEL SECURITY;
CREATE POLICY "compras_listas_select" ON compras_listas FOR SELECT
  USING (auth.uid() = user_id OR (es_compartida = true AND EXISTS (
    SELECT 1 FROM parejas p WHERE (p.user_a = auth.uid() AND p.user_b = compras_listas.user_id)
                                OR (p.user_b = auth.uid() AND p.user_a = compras_listas.user_id))));
CREATE POLICY "compras_listas_insert" ON compras_listas FOR INSERT WITH CHECK (auth.uid() = user_id);
CREATE POLICY "compras_listas_update" ON compras_listas FOR UPDATE
  USING (auth.uid() = user_id OR (es_compartida = true AND EXISTS (
    SELECT 1 FROM parejas p WHERE (p.user_a = auth.uid() AND p.user_b = compras_listas.user_id)
                                OR (p.user_b = auth.uid() AND p.user_a = compras_listas.user_id))));
CREATE POLICY "compras_listas_delete" ON compras_listas FOR DELETE USING (auth.uid() = user_id);

💑 SQL: Vínculo de pareja (ejecutar una sola vez) 
-- 1) Tabla de vínculo confirmado
CREATE TABLE parejas (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  user_a uuid REFERENCES auth.users NOT NULL,
  user_b uuid REFERENCES auth.users NOT NULL,
  email_a text,
  email_b text,
  created_at timestamptz DEFAULT now()
);
ALTER TABLE parejas ENABLE ROW LEVEL SECURITY;
CREATE POLICY "parejas_select" ON parejas FOR SELECT
  USING (auth.uid() = user_a OR auth.uid() = user_b);
CREATE POLICY "parejas_insert" ON parejas FOR INSERT
  WITH CHECK (
    auth.uid() = user_b
    AND EXISTS (SELECT 1 FROM pareja_solicitudes s
      WHERE s.from_user = parejas.user_a
        AND lower(s.to_email) = lower(auth.jwt()->>'email')
        AND s.estado = 'pendiente')
  );
CREATE POLICY "parejas_delete" ON parejas FOR DELETE
  USING (auth.uid() = user_a OR auth.uid() = user_b);

-- 2) Tabla de invitaciones por email
CREATE TABLE pareja_solicitudes (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  from_user uuid REFERENCES auth.users NOT NULL,
  from_email text,
  to_email text NOT NULL,
  estado text DEFAULT 'pendiente',
  created_at timestamptz DEFAULT now()
);
ALTER TABLE pareja_solicitudes ENABLE ROW LEVEL SECURITY;
CREATE POLICY "solicitudes_select" ON pareja_solicitudes FOR SELECT
  USING (auth.uid() = from_user OR lower(to_email) = lower(auth.jwt()->>'email'));
CREATE POLICY "solicitudes_insert" ON pareja_solicitudes FOR INSERT
  WITH CHECK (auth.uid() = from_user);
CREATE POLICY "solicitudes_update" ON pareja_solicitudes FOR UPDATE
  USING (auth.uid() = from_user OR lower(to_email) = lower(auth.jwt()->>'email'));
CREATE POLICY "solicitudes_delete" ON pareja_solicitudes FOR DELETE
  USING (auth.uid() = from_user);

-- 3) Reemplazar políticas de gastos e ingresos: de "cualquier usuario" a "solo tu pareja vinculada"
DO $$
DECLARE pol record;
BEGIN
  FOR pol IN SELECT policyname FROM pg_policies WHERE tablename = 'gastos' LOOP
    EXECUTE format('DROP POLICY %I ON gastos', pol.policyname);
  END LOOP;
  FOR pol IN SELECT policyname FROM pg_policies WHERE tablename = 'ingresos' LOOP
    EXECUTE format('DROP POLICY %I ON ingresos', pol.policyname);
  END LOOP;
END $$;

CREATE POLICY "gastos_select" ON gastos FOR SELECT USING (
  auth.uid() = user_id
  OR (es_compartido = true AND EXISTS (
    SELECT 1 FROM parejas p WHERE (p.user_a = auth.uid() AND p.user_b = gastos.user_id)
                                OR (p.user_b = auth.uid() AND p.user_a = gastos.user_id)))
);
CREATE POLICY "gastos_insert" ON gastos FOR INSERT WITH CHECK (auth.uid() = user_id);
CREATE POLICY "gastos_update" ON gastos FOR UPDATE USING (auth.uid() = user_id);
CREATE POLICY "gastos_delete" ON gastos FOR DELETE USING (auth.uid() = user_id);

✅ SQL: Estado de pago de gastos (ejecutar una sola vez) 
-- Agrega columna para marcar si un gasto ya fue pagado o sigue pendiente.
-- Default TRUE para no afectar retroactivamente el historial ya cargado.
ALTER TABLE gastos ADD COLUMN IF NOT EXISTS pagado boolean DEFAULT true;
💑 SQL: Permitir que la pareja edite gastos compartidos (ejecutar una sola vez) 
-- Hasta ahora, gastos_update solo permitía editar (tildar "Pagado", cambiar fecha, etc.)
-- al dueño del gasto (user_id). Esto bloqueaba a la pareja incluso en gastos es_compartido=true.
-- Mismo patrón que ya se aplicó a compras_items.
DROP POLICY IF EXISTS "gastos_update" ON gastos;
CREATE POLICY "gastos_update" ON gastos FOR UPDATE
USING (auth.uid() = user_id OR (es_compartido = true AND EXISTS (
SELECT 1 FROM parejas p WHERE (p.user_a = auth.uid() AND p.user_b = gastos.user_id)
OR (p.user_b = auth.uid() AND p.user_a = gastos.user_id))));
💑 SQL: Pago por parte en gastos compartidos (ejecutar una sola vez) 
-- Hasta ahora "pagado" era un solo booleano por gasto (todo o nada).
-- Estas dos columnas permiten que cada persona marque su propia parte como pagada.
-- Default TRUE para no afectar retroactivamente el historial ya cargado, y para
-- inicializarlas con el mismo valor que ya tenía "pagado" en cada fila existente.
ALTER TABLE gastos ADD COLUMN IF NOT EXISTS pagado_dueno boolean DEFAULT true;
ALTER TABLE gastos ADD COLUMN IF NOT EXISTS pagado_pareja boolean DEFAULT true;
UPDATE gastos SET pagado_dueno = pagado, pagado_pareja = pagado WHERE es_compartido = true;
